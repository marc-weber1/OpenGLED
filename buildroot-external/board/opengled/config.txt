dtparam=spi=on

max_framebuffers=2

# USB gadget mode: the data port becomes a USB ethernet device so a
# phone/laptop can ssh in (see rootfs-overlay/etc/init.d/S30usbgadget)
dtoverlay=dwc2,dr_mode=peripheral
